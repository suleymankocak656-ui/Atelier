    (function generateAtelierAppIcon() {
      try {
        var size = 180;
        var canvas = document.createElement('canvas');
        canvas.width = size;
        canvas.height = size;
        var ctx = canvas.getContext('2d');
        if (!ctx) return;

        function roundRectPath(c, x, y, w, h, r) {
          c.beginPath();
          c.moveTo(x + r, y);
          c.arcTo(x + w, y, x + w, y + h, r);
          c.arcTo(x + w, y + h, x, y + h, r);
          c.arcTo(x, y + h, x, y, r);
          c.arcTo(x, y, x + w, y, r);

        ctx.fillRect(0, 0, size, size);

        return 'other';
      }

      function matchesPalette(item, palette) {
        if (!palette || palette === 'Dengeli' || palette === 'Tümü') return true;
        const g = classifyColor(item.color);
        if (palette === 'All-Black') return g === 'black' || g === 'gray';
        if (palette === 'Toprak Tonları') return g === 'earth' || g === 'white' || g === 'black';
        if (palette === 'Yüksek Kontrast') return g === 'black' || g === 'white';
        if (palette === 'Old Money') return g === 'earth' || g === 'white' || g === 'blue' || g === 'black';
        return true;
      }

      function isLightHex(hex) {
        const h = (hex || '#888888').replace('#', '');
        if (h.length !== 6) return false;
        const r = parseInt(h.slice(0, 2), 16);
        const g = parseInt(h.slice(2, 4), 16);
        const b = parseInt(h.slice(4, 6), 16);
        return (0.299 * r + 0.587 * g + 0.114 * b) > 168;
      }

        if (item.image) {

        return `

              ${getCategorySvg(item.category)}
            </div>
          </div>
        `;
      }

      function getCleanPool(category, location) {
        return state.wardrobe.filter(i => {
          if (i.dirty) return false;
          if (i.category !== category) return false;
          if (location && location !== 'Tümü' && i.location !== location) return false;
          return true;
        });
      }

      function pickSlotItem(category, opts, excludeId) {
        let pool = getCleanPool(category, opts.location);
        if (!pool.length) pool = getCleanPool(category, 'Tümü');
        if (!pool.length) return null;
        if (excludeId && pool.length > 1) pool = pool.filter(i => i.id !== excludeId);

        const scored = pool.map(item => {
          let w = 10 + Math.random() * 6;
          if (item.occasions && item.occasions.includes(opts.occasion)) w += 15;
          if (matchesPalette(item, opts.palette)) w += 16;
          if (opts.weather === 'Soğuk' && item.season === 'Kış') w += 9;
          if (opts.weather === 'Güneşli' && item.season !== 'Kış') w += 8;
          if (opts.fitPref === 'Rahat' && ['Oversize', 'Baggy', 'Relaxed', 'Boxy'].includes(item.fit)) w += 13;
          if (opts.fitPref === 'Özenli' && ['Regular', 'Slim', 'Classic'].includes(item.fit)) w += 13;
          if (opts.fitPref === 'Rahat & Özenli' && ['Relaxed', 'Boxy', 'Regular', 'Classic'].includes(item.fit)) w += 11;
          return { item, w };
        }).sort((a, b) => b.w - a.w);

        const top = scored.slice(0, Math.min(2, scored.length));
        return top[Math.floor(Math.random() * top.length)].item;
      }

      function buildOutfit(opts) {
        const wObj = WEATHERS.find(w => w.id === opts.weather) || WEATHERS[1];
        const slots = ['Üst', 'Alt', 'Ayakkabı', 'Aksesuar', 'Parfüm'];
        if (wObj.outer) slots.splice(2, 0, 'Dış Giyim');

        const hero = opts.virtualItem || (opts.heroId ? state.wardrobe.find(i => i.id === opts.heroId && !i.dirty) : null);
        if (hero && hero.category === 'Dış Giyim' && !slots.includes('Dış Giyim')) {
          slots.splice(2, 0, 'Dış Giyim');
        }

        const out = {};
        slots.forEach(cat => {
          if (hero && hero.category === cat) {
            out[cat] = hero.id;
          } else if (opts.locks && opts.locks[cat] && opts.existing && opts.existing[cat]) {
            const lockedItem = getItemById(opts.existing[cat]);
            if (lockedItem && !lockedItem.dirty) {
              out[cat] = lockedItem.id;
            } else {
              const p = pickSlotItem(cat, opts, null);
              if (p) out[cat] = p.id;
            }
          } else {
            const p = pickSlotItem(cat, opts, null);
            if (p) out[cat] = p.id;
          }
        });
        return out;
      }

      function evaluateOutfit(outfit, occasion, weather) {
        const items = Object.values(outfit).map(getItemById).filter(Boolean);
        const top = getItemById(outfit['Üst']);
        const bot = getItemById(outfit['Alt']);
        const out = getItemById(outfit['Dış Giyim']);
        const sho = getItemById(outfit['Ayakkabı']);
        const acc = getItemById(outfit['Aksesuar']);
        const frg = getItemById(outfit['Parfüm']);

        // 1. Renk Uyumu
        const groups = new Set(items.filter(i => i.category !== 'Parfüm').map(i => classifyColor(i.color)));
        let colorScore = 9.4;
        let colorComment = 'Nötr ve dengeli renk geçişi; göz yormayan lüks bir palet.';
        if (groups.size === 1) {
          colorScore = 9.6;
          colorComment = 'Monokrom ton bütünlüğü çok keskin ve editoryal duruyor.';
        } else if (groups.size >= 4) {
          colorScore = 7.9;
          colorComment = 'Palet biraz dağılmış; bir parçayı siyah, krem veya griye çekersen bütünlük artar.';
        } else if (groups.has('earth') && (groups.has('black') || groups.has('white'))) {
          colorScore = 9.5;
          colorComment = 'Toprak tonları ile temel nötrlerin birleşimi zamansız bir kontrast kuruyor.';
        }

        // 2. Kalıp & Silüet Dengesi
        let fitScore = 9.0;
        let fitComment = 'Üst ve alt oranları dengeli, temiz bir günlük silüet.';
        if (bot && bot.fit === 'Baggy' && sho && sho.fit === 'Chunky') {
          fitScore = 9.7;
          fitComment = 'Baggy paça dökümü chunky tabanlı ayakkabıyla tam oturuyor.';
        } else if (bot && bot.fit === 'Baggy' && sho && sho.fit === 'Classic') {
          fitScore = 8.1;
          fitComment = 'Baggy pantolon altında klasik ince taban biraz zayıf kalabilir; chunky sneaker ile dene.';
        } else if (top && top.fit === 'Boxy' && bot && ['Baggy', 'Relaxed'].includes(bot.fit)) {
          fitScore = 9.5;
          fitComment = 'Boxy üst + geniş kesim alt bacak boyunu optik olarak uzatıyor.';
        } else if (top && top.fit === 'Oversize' && out && out.fit === 'Slim') {
          fitScore = 7.6;
          fitComment = 'Oversize üstün üstüne dar dış giyim omuzlarda potluk yapabilir.';
        }

        // 3. Metal & Aksesuar Uyumu
        let metalScore = 9.1;
        let metalComment = acc
          ? `${acc.metal || 'Minimal'} detaylı ${acc.name} kombine rafine bir son dokunuş veriyor.`
          : 'Saat veya ince bir kolye eklersen silüet daha tamamlanmış görünür.';
        if (acc && acc.metal === 'Gümüş' && (groups.has('black') || groups.has('gray') || groups.has('blue'))) {
          metalScore = 9.6;
          metalComment = 'Soğuk tonlu kıyafetlerle gümüş metal uyumu kusursuz çalışıyor.';
        }

        // 4. Parfüm Eşleşmesi
        let fragScore = 9.0;
        let fragComment = frg ? `${frg.name} seçilen ortamla uyumlu.` : 'Koku seçilmedi.';
        if (frg) {
          const isNight = occasion === 'Date' || occasion === 'Akşam Çıkışı' || weather === 'Soğuk' || weather === 'Serin / Rüzgarlı';
          if (isNight && frg.frag && (frg.frag.includes('Odunsu') || frg.frag.includes('Misk'))) {
            fragScore = 9.7;
            fragComment = `${frg.name}, ${weather.toLowerCase()} hava ve ${occasion} atmosferi için nokta atışı.`;
          } else if (!isNight && frg.frag && frg.frag.includes('Ferah')) {
            fragScore = 9.6;
            fragComment = `${frg.name} gün içi dinamizmine taze ve temiz bir imza katıyor.`;
          } else if (weather === 'Güneşli' && frg.frag && frg.frag.includes('Odunsu')) {
            fragScore = 8.2;
            fragComment = 'Sıcak havada ağır odunsu notalar yerine ferah narenciye veya temiz misk daha rahat taşınır.';
          }
        }

        const total = Number(((colorScore + fitScore + metalScore + fragScore) / 4).toFixed(1));
        const verdict = total >= 9.2 ? 'Kusursuz' : total >= 8.5 ? 'Çok İyi' : 'Geliştirilebilir';
        const summary = total >= 9.2
          ? 'Renk, kalıp ve detay uyumu tam bir Apple editoryal minimalizm dengesinde.'
          : total >= 8.5
          ? 'Güçlü ve güvenli bir kombin; ufak bir ayakkabı veya koku dokunuşuyla zirveye çıkar.'
          : 'Kalıp veya renk geçişinde küçük bir revizyonla çok daha keskin görünebilir.';

        return {
          total,
          verdict,
          summary,
          metrics: [
            { label: 'Renk Uyumu', score: colorScore, comment: colorComment },
            { label: 'Kalıp & Silüet Dengesi', score: fitScore, comment: fitComment },
            { label: 'Metal / Aksesuar Uyumu', score: metalScore, comment: metalComment },
            { label: 'Parfüm Eşleşmesi', score: fragScore, comment: fragComment }
          ]
        };
      }

      /* ------------------------------------------------------------------
         6. TURKISH NATURAL LANGUAGE PARSER (SPOTLIGHT BAR)
         ------------------------------------------------------------------ */
      function normalizeTr(str) {
        return (str || '').toLocaleLowerCase('tr-TR').replace(/[.,!?;:'"()]/g, ' ').replace(/\s+/g, ' ').trim();
      }

      function hasAny(text, words) {
        return words.some(w => text.includes(w));
      }

      function parseTurkishCommand(rawText) {
        const text = normalizeTr(rawText);
        const intents = {};

        // Occasion
        if (hasAny(text, ['date', 'randevu', 'romantik', 'buluşma'])) intents.occasion = 'Date';
        else if (hasAny(text, ['kahve', 'kafe', 'brunch', 'arkadaş'])) intents.occasion = 'Kahve';
        else if (hasAny(text, ['spor', 'gym', 'yürüyüş', 'koşu'])) intents.occasion = 'Spor / Rahat';
        else if (hasAny(text, ['akşam çıkışı', 'gece çıkışı', 'bar', 'kulüp', 'akşam yemeği', 'ofis', 'toplantı'])) intents.occasion = 'Akşam Çıkışı';
        else if (hasAny(text, ['okul', 'kampüs', 'ders', 'üniversite', 'günlük'])) intents.occasion = 'Günlük / Okul';

        // Weather
        if (hasAny(text, ['yağmur', 'ıslak', 'sağanak'])) intents.weather = 'Yağmurlu';
        else if (hasAny(text, ['soğuk', 'kış', 'ayaz', 'kar'])) intents.weather = 'Soğuk';
        else if (hasAny(text, ['güneş', 'sıcak', 'yaz'])) intents.weather = 'Güneşli';
        else if (hasAny(text, ['rüzgar', 'esinti', 'serin', 'bulut', 'akşam', 'gece'])) intents.weather = 'Serin / Rüzgarlı';

        // Palette
        if (hasAny(text, ['all-black', 'all black', 'monokrom', 'koyu ton', 'simsiyah', 'siyah kombin'])) intents.palette = 'All-Black';
        else if (hasAny(text, ['toprak', 'bej', 'krem ton', 'kahve ton', 'haki'])) intents.palette = 'Toprak Tonları';
        else if (hasAny(text, ['kontrast', 'siyah beyaz'])) intents.palette = 'Yüksek Kontrast';
        else if (hasAny(text, ['old money', 'minimal', 'klasik', 'özenli', 'şık'])) intents.palette = 'Old Money';

        // Fit preference
        const wantsRelaxed = hasAny(text, ['rahat', 'oversize', 'baggy', 'bol', 'salaş']);
        const wantsSharp = hasAny(text, ['özenli', 'şık', 'slim', 'keskin', 'toparlayıcı']);
        if (wantsRelaxed && wantsSharp) intents.fitPref = 'Rahat & Özenli';
        else if (wantsRelaxed) intents.fitPref = 'Rahat';
        else if (wantsSharp) intents.fitPref = 'Özenli';

        // Location
        if (hasAny(text, ['ev 2', 'ev2', 'ikinci ev'])) intents.location = 'Ev 2';
        else if (hasAny(text, ['ev 1', 'ev1', 'birinci ev'])) intents.location = 'Ev 1';

        // Hero piece detection from clean wardrobe
        const garmentWords = ['sneaker', 'loafer', 'bot', 'ceket', 'deri', 'trençkot', 'kaban', 'gömlek', 'oxford', 'hoodie', 'kazak', 'triko', 'tişört', 'pantolon', 'jean', 'denim', 'eşofman', 'baggy', 'kargo', 'saat', 'gözlük', 'kolye'];
        const colorWords = ['beyaz', 'siyah', 'krem', 'bej', 'antrasit', 'gri', 'lacivert', 'mavi', 'haki', 'kahverengi', 'taş', 'gümüş'];

        let bestHero = null;
        let bestScore = 0;
        state.wardrobe.filter(i => !i.dirty && (!intents.location || i.location === intents.location)).forEach(item => {
          const iName = normalizeTr(item.name);
          const iColor = normalizeTr(item.color);
          const iFit = normalizeTr(item.fit);
          let s = 0;
          garmentWords.forEach(gw => { if (text.includes(gw) && (iName.includes(gw) || iFit.includes(gw))) s += 6; });
          colorWords.forEach(cw => { if (text.includes(cw) && (iColor.includes(cw) || iName.includes(cw))) s += 4; });
          if (s >= 6 && s > bestScore) {
            bestScore = s;
            bestHero = item;
          }
        });
        if (bestHero) intents.heroItem = bestHero;

        return intents;
      }

      function executeNaturalLanguage(rawText) {
        const cleanText = (rawText || '').trim();
        if (!cleanText) {
          showToast('Lütfen nasıl bir kombin istediğini tek cümleyle yaz.');
          return;
        }
        const intents = parseTurkishCommand(cleanText);
        state.studio.virtualItem = null;
        state.studio.nlPrompt = cleanText;
        state.studio.nlIntents = {
          occasion: intents.occasion || null,
          weather: intents.weather || null,
          palette: intents.palette || null,
          fitPref: intents.fitPref || null,
          location: intents.location || null,
          heroName: intents.heroItem ? intents.heroItem.name : null
        };

        if (intents.occasion) state.studio.occasion = intents.occasion;
        if (intents.weather) state.studio.weather = intents.weather;
        state.studio.palette = intents.palette || 'Dengeli';
        state.studio.location = intents.location || 'Tümü';
        state.studio.fitPref = intents.fitPref || '';

        if (intents.heroItem) {
          state.studio.heroId = intents.heroItem.id;
          state.studio.locks = { [intents.heroItem.category]: true };
        } else {
          state.studio.heroId = '';
          state.studio.locks = {};
        }

        state.studio.outfit = buildOutfit({
          occasion: state.studio.occasion,
          weather: state.studio.weather,
          palette: state.studio.palette,
          location: state.studio.location,
          heroId: state.studio.heroId,
          fitPref: state.studio.fitPref,
          locks: state.studio.locks
        });

        const outItem = getItemById(state.studio.outfit['Dış Giyim']);
        const frgItem = getItemById(state.studio.outfit['Parfüm']);
        const noteParts = [];
        if (intents.heroItem) {
          noteParts.push(`"${intents.heroItem.name}" parçanı merkeze alarak ${state.studio.occasion} için dengeli bir silüet oluşturdum.`);
        } else if (intents.fitPref === 'Rahat & Özenli') {
          noteParts.push(`İstediğin rahat ama özenli havayı geniş kesim konforu ile temiz nötr tonları birleştirerek yakaladım.`);
        } else {
          noteParts.push(`"${cleanText}" talebine göre ${state.studio.occasion} ortamı ve ${state.studio.weather.toLowerCase()} hava için en uyumlu temiz parçaları seçtim.`);
        }
        if (outItem) noteParts.push(`Hava şartlarına karşı ${outItem.name} katmanı eklendi.`);
        if (frgItem) noteParts.push(`Koku imzası olarak ${frgItem.name} eşleştirildi.`);

        state.studio.nlNote = noteParts.join(' ');
        saveState();
        renderStudio();
        showToast('✦ Komutun çözümlendi ve kombin hazırlandı!');
      }

      function switchTab(tabId) {
        state.activeTab = tabId;
        document.querySelectorAll('.tab-section').forEach(s => s.classList.add('hidden'));

        if (target) target.classList.remove('hidden');

        document.querySelectorAll('[data-nav]').forEach(btn => {
          if (btn.classList.contains('nav-btn') || btn.classList.contains('dock-btn')) {

      function createTinderCard() {
        const occ = OCCASIONS[Math.floor(Math.random() * OCCASIONS.length)];
        const w = WEATHERS[Math.floor(Math.random() * WEATHERS.length)].id;
        const outfit = buildOutfit({ occasion: occ, weather: w, palette: 'Dengeli', location: 'Tümü' });
        return { occasion: occ, weather: w, outfit, evalRes: evaluateOutfit(outfit, occ, w) };
      }

      function attachTinderTouch() {
        const card = document.getElementById('active-tinder-card');
        if (!card) return;
        let startX = 0;
        card.addEventListener('touchstart', e => { startX = e.touches[0].clientX; }, { passive: true });
        card.addEventListener('touchend', e => {
          const dx = e.changedTouches[0].clientX - startX;
          if (dx > 60) swipeTinder('right');
          else if (dx < -60) swipeTinder('left');
        }, { passive: true });
      }

      function swipeTinder(dir) {
        const card = document.getElementById('active-tinder-card');
        if (!card) return;
        card.classList.add(dir === 'right' ? 'go-right' : 'go-left');
        if (dir === 'right' && state.tinderCard) {
          state.favorites.unshift({
            id: 'fav_' + Date.now(),
            occasion: state.tinderCard.occasion,
            weather: state.tinderCard.weather,
            score: state.tinderCard.evalRes.total,
            outfit: clone(state.tinderCard.outfit)
          });
          saveState();
          showToast('Kombin favorilerine eklendi ♥');
        }
        setTimeout(() => {
          state.tinderCard = createTinderCard();
          renderDiscover();
        }, 260);
      }

      /* ------------------------------------------------------------------
         15. PHOTO COLOR EXTRACTION (ADD ITEM MODAL)
         ------------------------------------------------------------------ */
      function rgbToHex(r, g, b) {
        return '#' + [r, g, b].map(x => x.toString(16).padStart(2, '0')).join('');
      }

      function guessColorName(r, g, b) {
        const max = Math.max(r, g, b), min = Math.min(r, g, b);
        const lum = 0.299 * r + 0.587 * g + 0.114 * b;
        const sat = max === 0 ? 0 : (max - min) / max;
        if (lum < 48) return 'Siyah';
        if (lum > 224 && sat < 0.16) return 'Beyaz';
        if (sat < 0.15) return lum < 115 ? 'Antrasit' : 'Gri';
        if (r > g && r > b) return lum > 180 ? 'Krem' : g > 95 ? 'Kahverengi' : 'Bordo';
        if (g >= r && g >= b) return 'Haki';
        if (b > r && b > g) return lum < 95 ? 'Lacivert' : 'Mavi';
        return 'Nötr';
      }

      document.getElementById('input-item-photo').addEventListener('change', e => {
        const file = e.target.files && e.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = ev => {
          uploadedPhoto = ev.target.result;
          document.getElementById('photo-preview').innerHTML = `<img src="${uploadedPhoto}" style="width:100%;height:100%;object-fit:cover;" />`;
          const img = new Image();
          img.onload = () => {
            const cv = document.getElementById('color-canvas');
            const ctx = cv.getContext('2d');
            cv.width = 24; cv.height = 24;
            ctx.drawImage(img, 0, 0, 24, 24);
            const data = ctx.getImageData(4, 4, 16, 16).data;
            let r = 0, g = 0, b = 0, n = 0;
            for (let i = 0; i < data.length; i += 4) { r += data[i]; g += data[i + 1]; b += data[i + 2]; n++; }
            r = Math.round(r / n); g = Math.round(g / n); b = Math.round(b / n);
            const hex = rgbToHex(r, g, b);
            const name = guessColorName(r, g, b);
            document.getElementById('add-hex').value = hex;
            document.getElementById('add-color').value = name;
            document.getElementById('photo-hint').textContent = `Otomatik algılandı: ${name} (${hex})`;
          };
          img.src = uploadedPhoto;
        };
        reader.readAsDataURL(file);
      });

        const pick = (cat, n) => state.wardrobe.filter(i => i.category === cat && !i.dirty).slice(0, n);
        const tops = pick('Üst', Math.min(days, Math.ceil(days * 0.7) + 1));
        const bots = pick('Alt', days <= 3 ? 2 : 3);
        const outs = weather === 'Güneşli' ? [] : pick('Dış Giyim', days >= 5 ? 2 : 1);
        const shos = pick('Ayakkabı', days <= 3 ? 1 : 2);
        const accs = pick('Aksesuar', 2);
        const frgs = pick('Parfüm', 1);
        const all = [...tops, ...bots, ...outs, ...shos, ...accs, ...frgs];
        const combos = tops.length * bots.length * Math.max(1, shos.length);
        state.suitcase = {
          days, weather, combos,
          items: all.map(i => ({ id: i.id, name: i.name, category: i.category, hex: i.hex, packed: false }))
        };
        saveState(); renderSuitcase();
        showToast(`Kapsül valiz hazır: ${all.length} parça ile ${combos} kombin!`);
      });

      // Sourcing & Budget Controls
      document.getElementById('sourcing-query-input').addEventListener('input', renderSourcingStores);
      document.getElementById('btn-copy-query').addEventListener('click', () => {
        const inp = document.getElementById('sourcing-query-input');
        inp.select();
        try {
          document.execCommand('copy');
          showToast('Arama sorgusu kopyalandı ✓');
        } catch (e) {
          showToast('Metni manuel kopyalayabilirsin.');
        }
      });

      document.getElementById('btn-open-add-wish').addEventListener('click', () => {
        document.getElementById('form-add-wish').reset();
        openModal('modal-add-wish');
      });

      document.getElementById('form-add-wish').addEventListener('submit', e => {
        e.preventDefault();
        const w = {
          id: 'w_' + Date.now(),
          name: document.getElementById('wish-name').value.trim(),
          category: document.getElementById('wish-cat').value,
          fit: document.getElementById('wish-fit').value,
          color: document.getElementById('wish-color').value.trim(),
          hex: document.getElementById('wish-hex').value,
          price: Number(document.getElementById('wish-price').value) || 0,
          tip: document.getElementById('wish-tip').value.trim()
        };
        state.wishlist.unshift(w);
        saveState();
        closeModal('modal-add-wish');
        renderSourcing();
        showToast(`"${w.name}" alışveriş listesine eklendi ✓`);
      });

      document.getElementById('btn-edit-budget').addEventListener('click', () => {
        document.getElementById('input-budget').value = state.budget;
        openModal('modal-budget');
      });
      document.getElementById('btn-save-budget').addEventListener('click', () => {
        const v = Number(document.getElementById('input-budget').value);
        if (v > 0) { state.budget = v; saveState(); renderSourcing(); }
        closeModal('modal-budget');
        showToast('Bütçe güncellendi ✓');
      });

      document.getElementById('btn-reset-data').addEventListener('click', () => {
        safeStorage.remove(STORAGE_KEY);
        seedDefaults();
        state.studio.virtualItem = null;
        state.studio.outfit = buildOutfit(state.studio);
        saveState();
        switchTab('studio');
        showToast('Tüm veriler varsayılana sıfırlandı.');
      });

      // Global Delegated Click Handler
      document.body.addEventListener('click', e => {
        const navEl = e.target.closest('[data-nav]');
        if (navEl) { switchTab(navEl.dataset.nav); return; }

        const closeEl = e.target.closest('[data-close-modal]');
        if (closeEl) { closeModal(closeEl.dataset.closeModal); return; }

        if (e.target.classList.contains('modal-backdrop')) {
          e.target.classList.add('hidden');
          return;
        }

        const nlChip = e.target.closest('.nl-example');
        if (nlChip) {
          document.getElementById('nl-input').value = nlChip.dataset.prompt;
          executeNaturalLanguage(nlChip.dataset.prompt);
          return;
        }

        const occBtn = e.target.closest('[data-set-occ]');
        if (occBtn) {
          state.studio.occasion = occBtn.dataset.setOcc;
          state.studio.outfit = buildOutfit(state.studio);
          saveState(); renderStudio(); return;
        }
        const wBtn = e.target.closest('[data-set-weather]');
        if (wBtn) {
          state.studio.weather = wBtn.dataset.setWeather;
          state.studio.outfit = buildOutfit(state.studio);
          saveState(); renderStudio(); return;
        }

        const lockBtn = e.target.closest('[data-lock-slot]');
        if (lockBtn) {
          const cat = lockBtn.dataset.lockSlot;
          state.studio.locks[cat] = !state.studio.locks[cat];
          saveState(); renderStudio(); return;
        }
        const swapBtn = e.target.closest('[data-swap-slot]');
        if (swapBtn) {
          const cat = swapBtn.dataset.swapSlot;
          if (state.studio.virtualItem && state.studio.virtualItem.category === cat) {
            showToast('Sanal deneme parçasını değiştirmek için denemeden çık.');
            return;
          }
          const next = pickSlotItem(cat, state.studio, state.studio.outfit[cat]);
          if (next) {
            state.studio.outfit[cat] = next.id;
            saveState(); renderStudio();
          } else {
            showToast('Bu kategoride başka temiz parça yok.');
          }
          return;
        }

        const emBtn = e.target.closest('[data-pick-emerg]');
        if (emBtn) {
          const opt = emergencyOptions[Number(emBtn.dataset.pickEmerg)];
          if (opt) {
            state.studio.virtualItem = null;
            state.studio.outfit = clone(opt.outfit);
            state.studio.locks = {};
            closeModal('modal-emergency');
            switchTab('studio');
            showToast('⚡ Garanti kombin Stüdyoya yüklendi!');
          }
          return;
        }

        const loadFav = e.target.closest('[data-load-fav]');
        if (loadFav) {
          const f = state.favorites.find(x => x.id === loadFav.dataset.loadFav);
          if (f) {
            state.studio.virtualItem = null;
            state.studio.occasion = f.occasion;
            state.studio.weather = f.weather || 'Serin / Rüzgarlı';
            state.studio.outfit = clone(f.outfit);
            state.studio.locks = {};
            switchTab('studio');
          }
          return;
        }
        const delFav = e.target.closest('[data-del-fav]');
        if (delFav) {
          state.favorites = state.favorites.filter(x => x.id !== delFav.dataset.delFav);
          saveState(); renderDiscover(); return;
        }

        const wSub = e.target.closest('[data-wsub]');
        if (wSub) {
          state.wardrobeSub = wSub.dataset.wsub;
          document.querySelectorAll('[data-wsub]').forEach(b => b.classList.toggle('active', b.dataset.wsub === state.wardrobeSub));
          ['all', 'laundry', 'care'].forEach(k => document.getElementById('wsub-' + k).classList.toggle('hidden', k !== state.wardrobeSub));
          renderWardrobe(); return;
        }
        const wCat = e.target.closest('[data-wcat]');
        if (wCat) { state.wardrobeCat = wCat.dataset.wcat; renderWardrobe(); return; }

        const togDirty = e.target.closest('[data-toggle-dirty]');
        if (togDirty) {
          const it = state.wardrobe.find(x => x.id === togDirty.dataset.toggleDirty);
          if (it) {
            it.dirty = !it.dirty;
            saveState(); renderWardrobe();
            showToast(it.dirty ? `${it.name} kirli sepetine atıldı.` : `${it.name} temize alındı.`);
          }
          return;
        }
        const delItem = e.target.closest('[data-del-item]');
        if (delItem) {
          state.wardrobe = state.wardrobe.filter(x => x.id !== delItem.dataset.delItem);
          saveState(); renderWardrobe(); showToast('Parça silindi.'); return;
        }
        const cleanBtn = e.target.closest('[data-mark-cleaned]');
        if (cleanBtn) {
          const it = state.wardrobe.find(x => x.id === cleanBtn.dataset.markCleaned);
          if (it) { it.cleanedAtWear = it.wearCount; saveState(); renderWardrobe(); showToast('Temizlik kaydedildi ✓'); }
          return;
        }
        const heroBtn = e.target.closest('[data-make-hero]');
        if (heroBtn) {
          const it = state.wardrobe.find(x => x.id === heroBtn.dataset.makeHero);
          if (!it) return;
          if (it.dirty) { showToast('Bu parça kirli sepetinde, önce temize al.'); return; }
          state.studio.virtualItem = null;
          state.studio.heroId = it.id;
          state.studio.location = 'Tümü';
          state.studio.locks = { [it.category]: true };
          state.studio.outfit = buildOutfit(state.studio);
          saveState(); switchTab('studio');
          showToast(`"${it.name}" odak parça olarak seçildi!`);
          return;
        }

        const pSub = e.target.closest('[data-psub]');
        if (pSub) {
          state.plannerSub = pSub.dataset.psub;
          document.querySelectorAll('[data-psub]').forEach(b => b.classList.toggle('active', b.dataset.psub === state.plannerSub));
          ['weekly', 'suitcase'].forEach(k => document.getElementById('psub-' + k).classList.toggle('hidden', k !== state.plannerSub));
          renderPlanner(); return;
        }
        const assignStudio = e.target.closest('[data-assign-studio]');
        if (assignStudio) {
          const day = assignStudio.dataset.assignStudio;
          const occ = (state.weeklyPlan[day] && state.weeklyPlan[day].occasion) || state.studio.occasion;
          const ev = evaluateOutfit(state.studio.outfit, occ, state.studio.weather);
          state.weeklyPlan[day] = { occasion: occ, outfit: clone(state.studio.outfit), score: ev.total };
          saveState(); renderPlanner(); showToast(`${day} gününe Stüdyo kombini atandı ✓`); return;
        }
        const assignRand = e.target.closest('[data-assign-random]');
        if (assignRand) {
          const day = assignRand.dataset.assignRandom;
          const occ = (state.weeklyPlan[day] && state.weeklyPlan[day].occasion) || 'Günlük / Okul';
          const outfit = buildOutfit({ occasion: occ, weather: state.studio.weather, palette: 'Dengeli', location: 'Tümü' });
          state.weeklyPlan[day] = { occasion: occ, outfit, score: evaluateOutfit(outfit, occ, state.studio.weather).total };
          saveState(); renderPlanner(); return;
        }
        const packItem = e.target.closest('[data-toggle-pack]');
        if (packItem && state.suitcase) {
          const found = state.suitcase.items.find(x => x.id === packItem.dataset.togglePack);
          if (found) { found.packed = !found.packed; saveState(); renderSuitcase(); }
          return;
        }

        const rTry = e.target.closest('[data-radar-try]');
        if (rTry) { const r = computeMissingRadar()[Number(rTry.dataset.radarTry)]; if (r) startVirtualTryOn(r); return; }
        const rSrc = e.target.closest('[data-radar-source]');
        if (rSrc) { const r = computeMissingRadar()[Number(rSrc.dataset.radarSource)]; if (r) openSourcingModal(r); return; }
        const rWish = e.target.closest('[data-radar-wish]');
        if (rWish) {
          const r = computeMissingRadar()[Number(rWish.dataset.radarWish)];
          if (r) {
            if (state.wishlist.some(x => x.name === r.name)) { showToast('Bu parça zaten listende mevcut.'); return; }
            state.wishlist.unshift({ id: 'w_' + Date.now(), name: r.name, category: r.category, fit: r.fit, color: r.color, hex: r.hex, price: r.price, tip: r.tip });
            saveState(); updateBadges();
            if (state.activeTab === 'sourcing') renderSourcing();
            showToast(`"${r.name}" alınacaklara eklendi ✓`);
          }
          return;
        }

        const wTry = e.target.closest('[data-wish-try]');
        if (wTry) { const w = state.wishlist.find(x => x.id === wTry.dataset.wishTry); if (w) startVirtualTryOn(w); return; }
        const wSrc = e.target.closest('[data-wish-source]');
        if (wSrc) { const w = state.wishlist.find(x => x.id === wSrc.dataset.wishSource); if (w) openSourcingModal(w); return; }
        const wDel = e.target.closest('[data-del-wish]');
        if (wDel) {
          state.wishlist = state.wishlist.filter(x => x.id !== wDel.dataset.delWish);
          saveState(); renderSourcing(); showToast('Listeden çıkarıldı.'); return;
        }
        const wBuy = e.target.closest('[data-wish-buy]');
        if (wBuy) {
          const w = state.wishlist.find(x => x.id === wBuy.dataset.wishBuy);
          if (w) {
            state.wardrobe.unshift({
              id: 'it_' + Date.now(),
              name: w.name, category: w.category, fit: w.fit || 'Regular',
              color: w.color || 'Nötr', hex: w.hex || '#141413',
              location: 'Ev 1', season: 'Dört Mevsim',
              occasions: ['Günlük / Okul', 'Kahve', 'Date'],
              price: Number(w.price) || 0, wearCount: 0, lastWornDays: 0, dirty: false,
              care: w.tip || '30°C hassas yıkama.'
            });
            state.budgetSpent = (Number(state.budgetSpent) || 0) + (Number(w.price) || 0);
            state.wishlist = state.wishlist.filter(x => x.id !== w.id);
            if (state.studio.virtualItem && state.studio.virtualItem.name === w.name) state.studio.virtualItem = null;
            saveState(); renderSourcing();
            showToast(`🎉 "${w.name}" dolabına aktarıldı!`);
          }
          return;
        }
      });

      document.body.addEventListener('change', e => {
        if (e.target.matches('[data-day-occ]')) {
          const day = e.target.dataset.dayOcc;
          const occ = e.target.value;
          const outfit = buildOutfit({ occasion: occ, weather: state.studio.weather, palette: 'Dengeli', location: 'Tümü' });
          state.weeklyPlan[day] = { occasion: occ, outfit, score: evaluateOutfit(outfit, occ, state.studio.weather).total };
          saveState(); renderPlanner();
        }
      });

      /* ------------------------------------------------------------------
         17. BOOTSTRAP
         ------------------------------------------------------------------ */
      loadState();
      if (!state.studio.outfit || !Object.keys(state.studio.outfit).length) {
        state.studio.outfit = buildOutfit(state.studio);
      }
      switchTab(state.activeTab || 'studio');
    });
  </script>
